```python
    # 延迟指标
    llm_call_latency_ms: float = 0
    tool_call_latency_ms: float = 0
    total_execution_latency_ms: float = 0

    # 吞吐量指标
    requests_per_second: float = 0

    # 资源指标
    cpu_usage_pct: float = 0
    memory_usage_mb: float = 0
    memory_usage_pct: float = 0

    # Token指标
    prompt_tokens: int = 0
    completion_tokens: int = 0
    total_tokens: int = 0

    # 错误指标
    error_count: int = 0
    retry_count: int = 0

class MetricsCollector:
    def __init__(self):
        self.metrics_buffer: List[PerformanceMetrics] = []
        self.process = psutil.Process()

    def collect_snapshot(self) -> PerformanceMetrics:
        """采集当前性能快照"""
        return PerformanceMetrics(
            timestamp=datetime.now(),

            # 内存指标
            memory_usage_mb=self.process.memory_info().rss / 1024 / 1024,
            memory_usage_pct=self.process.memory_percent(),

            # CPU指标
            cpu_usage_pct=self.process.cpu_percent(interval=0.1),

            # IO指标
            io_counters=self.process.io_counters() if hasattr(self.process, 'io_counters') else None
        )

    async def start_monitoring(
        self,
        interval_seconds: float = 1.0
    ):
        """开始持续监控"""
        while True:
            metrics = self.collect_snapshot()
            self.metrics_buffer.append(metrics)

            # 保持缓冲区大小
            if len(self.metrics_buffer) > 10000:
                self.metrics_buffer.pop(0)

            await asyncio.sleep(interval_seconds)
```

```python
        self,
        name: str,
        attributes: Dict[str, str] = None

    async def trace_task_execution(
        self,
        task: Task,
        agent_id: str
    ):
        """追踪任务执行"""
        with self.create_span(
            "task_execution",
            {"task_id": task.id, "agent_id": agent_id}

            span.set_attribute("subtask_count", len(subtasks))

                results = await self.execute_subtasks_parallel(subtasks)

            # 追踪结果聚合
            with self.create_span("result_aggregation"):
                final_result = self.aggregate_results(results)

            span.set_attribute("success", final_result.is_success)
```

### 3.1.1 缓存优化

```python
import hashlib
import json
from typing import Any, Optional

class LLMResponseCache:
    """LLM响应缓存"""
    def __init__(self, redis_client=None):
        self.cache: Dict[str, CacheEntry] = {}
        self.redis = redis_client
        self.hit_count = 0
        self.miss_count = 0

    def _generate_cache_key(
        self,
        model: str,
        messages: List[Dict],
        parameters: Dict[str, Any]
    ) -> str:
        """生成缓存键"""
        content = json.dumps({
            "model": model,
            "messages": messages,
            "parameters": parameters
        }, sort_keys=True)

        return hashlib.sha256(content.encode()).hexdigest()

    async def get_cached_response(
        self,
        model: str,
        messages: List[Dict],
        parameters: Dict[str, Any]
    ) -> Optional[str]:
        """获取缓存的响应"""
        cache_key = self._generate_cache_key(model, messages, parameters)

        # 优先从Redis获取
        if self.redis:
            cached = await self.redis.get(cache_key)
            if cached:
                self.hit_count += 1
                return cached.decode()

        # 从内存缓存获取
        if cache_key in self.cache:
            entry = self.cache[cache_key]

            if entry.is_valid():
                self.hit_count += 1
                return entry.response

        self.miss_count += 1
        return None

    async def cache_response(
        self,
        model: str,
        messages: List[Dict],
        parameters: Dict[str, Any],
        response: str,
        ttl_seconds: int = 3600
    ):
        """缓存响应"""
        cache_key = self._generate_cache_key(model, messages, parameters)

        entry = CacheEntry(
            response=response,
            created_at=datetime.now(),
            ttl_seconds=ttl_seconds
        )

        # 存入内存缓存
        self.cache[cache_key] = entry

        # 存入Redis
        if self.redis:
            await self.redis.setex(
                cache_key,
                ttl_seconds,
                response
            )

    def get_hit_rate(self) -> float:
        """获取缓存命中率"""
        total = self.hit_count + self.miss_count
        return self.hit_count / total if total > 0 else 0.0
```

```python
class AsyncLLMOptimizer:
    """异步LLM调用优化"""
    def __init__(self, max_concurrent: int = 10):
        self.semaphore = asyncio.Semaphore(max_concurrent)
        self.batch_size = 10
        self.pending_requests: List[PendingRequest] = []

    async def batch_complete(
        self,
        requests: List[LLMRequest]
    ) -> List[LLMResponse]:
        """批量LLM调用"""
        # 将请求分组
        batches = [
            requests[i:i + self.batch_size]
            for i in range(0, len(requests), self.batch_size)
        ]

        all_results = []

        for batch in batches:
            # 并发执行批次内请求
            batch_results = await asyncio.gather(
                *[self._complete_with_semaphore(req) for req in batch],
                return_exceptions=True
            )
            all_results.extend(batch_results)

        return all_results

    async def _complete_with_semaphore(
        self,
        request: LLMRequest
    ) -> LLMResponse:
        """使用信号量控制并发"""
        async with self.semaphore:
            start_time = time.time()

            try:
                response = await self.llm.acomplete(
                    messages=request.messages,
                    model=request.model,
                    **request.parameters
                )

                return LLMResponse(
                    success=True,
                    content=response.content,
                    latency_ms=(time.time() - start_time) * 1000
                )

            except Exception as e:
                return LLMResponse(
                    success=False,
                    error=str(e),
                    latency_ms=(time.time() - start_time) * 1000
                )
```

```python
import tracemalloc
import gc

class MemoryLeakDetector:
    """内存泄漏检测器"""
    def __init__(self):
        self.snapshots: List[MemorySnapshot] = []
        self.leak_suspects: List[LeakSuspect] = []
        tracemalloc.start()

    def take_snapshot(self) -> MemorySnapshot:
        """拍摄内存快照"""
        snapshot = tracemalloc.take_snapshot()

        # 获取统计信息
        stats = snapshot.statistics('lineno')

        top_stats = [
            Statistic(
                filename=stat.traceback[0].filename,
                line_number=stat.traceback[0].lineno,
                size_kb=stat.size / 1024,
                count=stat.count
            )
            for stat in stats[:10]
        ]

        return MemorySnapshot(
            timestamp=datetime.now(),
            total_memory_mb=tracemalloc.get_traced_memory()[0] / 1024 / 1024,
            top_stats=top_stats
        )

    def detect_leak(self) -> LeakDetectionResult:
        """检测内存泄漏"""
        if len(self.snapshots) < 2:
            return LeakDetectionResult(is_leaking=False)

        # 比较最近的快照
        latest = self.snapshots[-1]
        previous = self.snapshots[-2]

        memory_growth = latest.total_memory_mb - previous.total_memory_mb

        # 如果内存增长超过阈值，标记为泄漏
        if memory_growth > 10:  # 10MB阈值
            suspected_sources = self._find_leak_sources(
                latest, previous
            )

            return LeakDetectionResult(
                is_leaking=True,
                growth_mb=memory_growth,
                suspected_sources=suspected_sources
            )

        return LeakDetectionResult(is_leaking=False)

    def _find_leak_sources(
        self,
        latest: MemorySnapshot,
        previous: MemorySnapshot
    ) -> List[str]:
        """查找泄漏源"""
        suspects = []

        for new_stat in latest.top_stats:
            for old_stat in previous.top_stats:
                if (new_stat.filename == old_stat.filename and
                    new_stat.line_number == old_stat.line_number):

                    growth = new_stat.size_kb - old_stat.size_kb

                    if growth > 100:  # 100KB增长
                        suspects.append(
                            f"{new_stat.filename}:{new_stat.line_number} "
                            f"(+{growth:.1f}KB)"
                        )

        return suspects[:5]
```

## 5.1 任务并行化

### 5.1.1 依赖感知的任务调度

```python
class DependencyAwareScheduler:
    """依赖感知的任务调度器"""
    def __init__(self, max_workers: int = 5):
        self.max_workers = max_workers
        self.executor = ThreadPoolExecutor(max_workers=max_workers)

    async def schedule_tasks(
        self,
        task_tree: TaskTree
    ) -> List[TaskResult]:
        # 1. 构建依赖图
        dependency_graph = self._build_dependency_graph(task_tree)

        # 2. 获取执行批次
        execution_batches = self._get_execution_batches(dependency_graph)

        all_results = {}

        # 3. 按批次执行
        for batch in execution_batches:
            batch_tasks = [
                task_tree.get_task(task_id)
                for task_id in batch
            ]

            # 并行执行批次内任务
            batch_results = await asyncio.gather(
                *[self._execute_task(task) for task in batch_tasks],
                return_exceptions=True
            )

            # 收集结果
            for task, result in zip(batch_tasks, batch_results):
                all_results[task.id] = result

        return list(all_results.values())

    def _get_execution_batches(
        self,
        dependency_graph: DiGraph
    ) -> List[List[str]]:
        """获取可并行执行的批次"""
        in_degree = {
            node: dependency_graph.in_degree(node)
            for node in dependency_graph.nodes
        }

        batches = []

        while True:
            # 找出入度为0的节点
            ready = [
                node for node, degree in in_degree.items()
                if degree == 0
            ]

            if not ready:
                break

            batches.append(ready)

            # 更新入度
            for node in ready:
                in_degree[node] = -1  # 标记为已处理

                for successor in dependency_graph.successors(node):
                    in_degree[successor] -= 1

        return batches
```

## 6.1 Token使用优化

### 6.1.1 Prompt压缩

```python
class PromptCompressor:
    """Prompt压缩器"""
    def __init__(self, llm: LLM):
        self.llm = llm

    async def compress_messages(
        self,
        messages: List[Dict[str, str]],
        max_tokens: int = 4000
    ) -> List[Dict[str, str]]:
        """压缩消息列表"""
        current_tokens = await self.count_tokens(messages)

        if current_tokens <= max_tokens:
            return messages

        # 保留系统消息和最新消息
        system_msg = None
        recent_messages = []

        for msg in messages:
            if msg["role"] == "system":
                system_msg = msg
            else:
                recent_messages.append(msg)

        # 从最新消息开始保留
        compressed = []
        total_tokens = 0

        for msg in reversed(recent_messages):
            msg_tokens = await self.count_tokens([msg])

            if total_tokens + msg_tokens > max_tokens - 500:  # 留出空间
                break

            compressed.insert(0, msg)
            total_tokens += msg_tokens

        # 如果有系统消息，放在最前面
        if system_msg:
            compressed.insert(0, system_msg)

        return compressed

    async def count_tokens(
        self,
        messages: List[Dict[str, str]]
    ) -> int:
        """计算token数量"""
        # 使用tiktoken计算
        encoding = tiktoken.get_encoding("cl100k_base")

        total = 0
        for msg in messages:
            total += len(encoding.encode(msg["content"]))

        return total
```
