    """
    动态优先级调整器
    根据上下文动态调整优先级
    """

    def __init__(self):
        self.time_sensitivity = TimeSensitivityCalculator()

    def adjust(self, base_priority, threat, current_context):
        """动态调整优先级"""
        adjustments = []

        time_adj = self.time_sensitivity.calculate(threat, current_context)
        adjustments.append(time_adj)

        context_adj = self.context_analyzer.analyze(threat, current_context)
        adjustments.append(context_adj)

        adjusted_priority = base_priority

        for adj in adjustments:
            if adj['type'] == 'boost':
                adjusted_priority *= (1 + adj['factor'])
            elif adj['type'] == 'reduce':
                adjusted_priority *= (1 - adj['factor'])

        return min(10.0, max(0.0, adjusted_priority))

class MultiDimensionalRiskQuantifier:

            score = dimension.evaluate(threat_data, asset_context)
            dimension_scores[dim_name] = score

        fused_risk = self.fusion_model.fuse(dimension_scores)

        return {
            'dimension_scores': dimension_scores,

class SeverityDimension:
    """
    严重性维度评估
    """

    def __init__(self, config):
        self.cvss_weight = config.get('cvss_weight', 0.4)
        self.business_impact_weight = config.get('business_impact_weight', 0.3)
        self.data_weight = config.get('data_weight', 0.3)

    def evaluate(self, threat_data, context):
        """评估严重性"""
        cvss_score = self._extract_cvss(threat_data)

        business_impact = self._evaluate_business_impact(
            threat_data, context
        )

        data_impact = self._evaluate_data_impact(
            threat_data, context
        )

        return (
            self.cvss_weight * cvss_score +
            self.business_impact_weight * business_impact +
            self.data_weight * data_impact
        )

    def _extract_cvss(self, threat_data):
        """提取CVSS评分"""
        if 'cvss_vector' in threat_data:
            return self._parse_cvss_vector(threat_data['cvss_vector'])

        return threat_data.get('cvss_base_score', 5.0)

    def _evaluate_business_impact(self, threat_data, context):
        """评估业务影响"""
        business_factors = {
            'revenue_impact': context.get('revenue_impact', 0),
            'operational_disruption': context.get('disruption_hours', 0) / 24,
            'reputation_damage': context.get('reputation_score', 0.5)
        }

        return np.mean(list(business_factors.values()))

class MLPriorityPredictor:
    """
    机器学习优先级预测器
    使用ML模型预测威胁优先级
    """

    def __init__(self, config):
        self.feature_extractor = PriorityFeatureExtractor()
        self.models = {
            'gradient_boosting': GradientBoostingPredictor(config['gb']),
            'neural_network': NeuralNetworkPredictor(config['nn']),
            'ensemble': EnsemblePredictor(config['ensemble'])
        }
        self.calibrator = ProbabilityCalibrator()

    def predict(self, threat_data, context):
        """预测优先级"""
        features = self.feature_extractor.extract(threat_data, context)

        predictions = {}
        for model_name, model in self.models.items():
            pred = model.predict(features)
            predictions[model_name] = pred

        ensemble_pred = self._ensemble_predictions(predictions)

        calibrated_pred = self.calibrator.calibrate(ensemble_pred)

        return {
            'priority_score': calibrated_pred,
            'priority_level': self._score_to_level(calibrated_pred),
            'model_predictions': predictions,
            'confidence': self._calculate_confidence(predictions)
        }

    def _ensemble_predictions(self, predictions):
        """集成预测"""
        weights = {
            'gradient_boosting': 0.35,
            'neural_network': 0.30,
            'ensemble': 0.35
        }

        ensemble_score = 0
        for model_name, pred in predictions.items():
            ensemble_score += weights[model_name] * pred

        return ensemble_score

class RealTimePriorityPipeline:
    """
    实时优先级处理管道
    处理海量威胁的优先级排序
    """

    def __init__(self, config):
        self.kafka_consumer = KafkaThreatConsumer(config['kafka'])
        self.feature_store = RedisFeatureStore(config['redis'])
        self.priority_processor = MLPriorityPredictor(config['ml'])
        self.alert_publisher = AlertPublisher(config['alerting'])
        self.metrics_collector = PipelineMetricsCollector()

    def process(self):
        """处理威胁流"""
        threats = self.kafka_consumer.consume(batch_size=100)

        for threat in threats:
            with self.metrics_collector.track_processing_time():
                processed = self._process_threat(threat)

                if processed['priority_level'] in ['critical', 'high']:
                    self.alert_publisher.publish(processed)

        self.metrics_collector.record_batch_metrics(threats)

    def _process_threat(self, threat):
        """处理单个威胁"""
        context = self.feature_store.get_context(threat['entity_id'])

        priority_result = self.priority_processor.predict(
            threat, context
        )

        enriched_threat = {
            **threat,
            **priority_result,
            'processing_timestamp': datetime.now().isoformat()
        }

        return enriched_threat


class KafkaThreatConsumer:
    """
    Kafka威胁消费者
    """

    def __init__(self, config):
        self.config = config
        self.consumer = KafkaConsumer(
            config['topic'],
            bootstrap_servers=config['bootstrap_servers'],
            group_id=config['consumer_group'],

        """消费威胁消息"""
        messages = self.consumer.poll(timeout_ms=1000, max_records=batch_size)

        threats = []
        for tp, records in messages.items():
            for record in records:
                threats.append(record.value)

        return threats

class PriorityQueueManager:
    """
    优先级队列管理器
    管理不同优先级的威胁队列
    """

    def __init__(self, config):
        self.queues = {
            'critical': PriorityQueue(maxsize=1000),
            'high': PriorityQueue(maxsize=5000),
            'medium': PriorityQueue(maxsize=20000),
            'low': PriorityQueue(maxsize=100000)
        }
        self.heap_manager = PriorityHeapManager()
        self.deduplicator = AlertDeduplicator()

    def enqueue(self, threat):
        """入队威胁"""
        if self.deduplicator.is_duplicate(threat):
            return False

        priority_level = threat['priority_level']
        self.queues[priority_level].put(threat)

        self.heap_manager.insert(threat)

        return True

    def dequeue(self, analyst_id):
        """出队威胁"""
        for queue_name in ['critical', 'high', 'medium', 'low']:
            queue = self.queues[queue_name]

            if not queue.empty():
                threat = queue.get()

                self._assign_to_analyst(threat, analyst_id)

                return threat

        return None

    def rebalance(self):
        """重平衡队列"""
        for queue_name, queue in self.queues.items():
            if queue.qsize() > self._get_max_size(queue_name) * 1.5:
                self._overflow_to_lower_queues(queue_name, queue)


class PriorityHeapManager:
    """
    优先级堆管理器
    使用堆结构高效管理优先级
    """

    def __init__(self):
        self.heap = []

    def insert(self, threat):
        """插入威胁"""
        heapq.heappush(self.heap, (
            -threat['priority_score'],
            threat['threat_id'],
            threat
        ))

    def extract_max(self):
        """提取最高优先级"""
        if not self.heap:
            return None

        return heapq.heappop(self.heap)[2]

    def peek(self):
        """查看最高优先级"""
        if not self.heap:
            return None

        return self.heap[0][2]
```

class ThresholdAutoTuner:
    """
    阈值自动调优器
    根据运营反馈自动调整优先级阈值
    """

    def __init__(self, config):
        self.target_precision = config.get('target_precision', 0.9)
        self.target_recall = config.get('target_recall', 0.85)
        self.optimizer = ThresholdOptimizer()

    def tune(self, historical_data, feedback):
        """调优阈值"""
        precision_by_level = self._calculate_precision_by_level(
            historical_data, feedback
        )

        recall_by_level = self._calculate_recall_by_level(
            historical_data, feedback
        )

        tuned_thresholds = {}

        for level in ['critical', 'high', 'medium', 'low']:
            current = self._get_current_threshold(level)

            if precision_by_level[level] < self.target_precision:
                adjusted = self.optimizer.increase_threshold(
                    current, precision_by_level[level], self.target_precision
                )
            elif recall_by_level[level] < self.target_recall:
                adjusted = self.optimizer.decrease_threshold(
                    current, recall_by_level[level], self.target_recall
                )
            else:
                adjusted = current

            tuned_thresholds[level] = adjusted

        return tuned_thresholds

    def _calculate_precision_by_level(self, historical, feedback):
        """计算各层级精确率"""
        precision = {}

        for level in ['critical', 'high', 'medium', 'low']:
            level_preds = [h for h in historical if h['predicted_level'] == level]
            level_feedback = [f for f in feedback if f['threat_id'] in [h['id'] for h in level_preds]]

            if level_feedback:
                true_positives = sum(1 for f in level_feedback if f['confirmed'])
                precision[level] = true_positives / len(level_feedback)
            else:
                precision[level] = 1.0

        return precision


class ThresholdOptimizer:
    """
    阈值优化器
    使用优化算法调整阈值
    """

    def __init__(self):
        self.min_threshold = 0.0
        self.max_threshold = 10.0

    def increase_threshold(self, current, current_precision, target):
        """提高阈值"""
        step = 0.1

        while current_precision < target and current < self.max_threshold:
            current += step
            current_precision = self._estimate_precision_at_threshold(current)

        return min(current, self.max_threshold)

    def decrease_threshold(self, current, current_recall, target):
        """降低阈值"""
        step = 0.1

        while current_recall < target and current > self.min_threshold:
            current -= step
            current_recall = self._estimate_recall_at_threshold(current)

        return max(current, self.min_threshold)
```

                confirmed_threats,

        """运行A/B测试"""

            'treatment': 'new_priority_model',
            'control': 'current_model',
            'duration': '7d',

            contributions.items(),
            key=lambda x: abs(x[1]),
            reverse=True
        )

        for factor, contribution in sorted_contributions[:5]:
            chain.append({
                'factor': factor,
                'contribution': contribution,
                'evidence': self._get_evidence(factor, threat)
            })

        return chain

class PrioritySystemEvaluator:
    """
    优先级系统评估器
    评估优先级排序系统的效果
    """

    def __init__(self):
        self.metrics_calculator = MetricsCalculator()

    def evaluate(self, predictions, ground_truth):
        """评估系统性能"""
        metrics = {
            'precision_at_k': self._calculate_precision_at_k(
                predictions, ground_truth, k=10
            ),
            'recall_at_k': self._calculate_recall_at_k(
                predictions, ground_truth, k=10
            ),
            'mrr': self._calculate_mean_reciprocal_rank(
                predictions, ground_truth
            ),
            'ndcg': self._calculate_ndcg(
                predictions, ground_truth
            )
        }

        metrics['overall_score'] = np.mean([
            metrics['precision_at_k'],
            metrics['recall_at_k'],
            metrics['mrr'],
            metrics['ndcg']
        ])

        return metrics

    def _calculate_precision_at_k(self, predictions, ground_truth, k):
        """计算K位置精确率"""
        top_k_predictions = predictions[:k]
        relevant_count = sum(
            1 for pred in top_k_predictions
            if pred['threat_id'] in ground_truth
        )

        return relevant_count / k

    def _calculate_ndcg(self, predictions, ground_truth):
        """计算NDCG"""
        dcg = self._calculate_dcg(predictions, ground_truth)
        idcg = self._calculate_dcg(
            sorted(ground_truth, key=lambda x: x['relevance'], reverse=True),
            ground_truth
        )

        return dcg / idcg if idcg > 0 else 0

class LoadBalancer:
    """
    负载均衡器
    """

    def __init__(self):
        self.strategy = 'round_robin'
        self.current_index = 0

    def select(self, instances):
        """选择实例"""
        if self.strategy == 'round_robin':
            selected = instances[self.current_index % len(instances)]
            self.current_index += 1
            return selected
        elif self.strategy == 'least_connections':
            return min(instances, key=lambda x: x.active_connections)
